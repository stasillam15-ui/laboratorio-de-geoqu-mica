    <script>
        // --- 1. Estado y Datos ---
        // Colores fijos para los cursos y docentes para mantener consistencia visual
        const COLORES_CURSOS = {
            "Geo metalurgia": { bg: "rgba(59, 130, 246, 0.7)", border: "rgba(59, 130, 246, 1)" },       // Azul
            "Aguas Superficiales": { bg: "rgba(16, 185, 129, 0.7)", border: "rgba(16, 185, 129, 1)" },  // Verde
            "No Metálicos": { bg: "rgba(245, 158, 11, 0.7)", border: "rgba(245, 158, 11, 1)" },         // Ámbar
            "Sedimentos": { bg: "rgba(139, 92, 246, 0.7)", border: "rgba(139, 92, 246, 1)" }            // Morado
        };

        const COLORES_DOCENTES = {
            "Dr. Wilder Chuquiruna": "rgba(59, 130, 246, 1)",
            "M.Sc. Sandra Medina Paredes": "rgba(16, 185, 129, 1)",
            "M.Sc. Paul Aguilar": "rgba(245, 158, 11, 1)",
            "M.Sc. Zoraida Remires": "rgba(139, 92, 246, 1)"
        };

        // Estado inicial de los registros
        let registrosLaboratorio = [
            { id: 1, docente: "Dr. Wilder Chuquiruna", curso: "Geo metalurgia", fecha: "2026-08-03", horas: 6 },
            { id: 2, docente: "M.Sc. Sandra Medina Paredes", curso: "Aguas Superficiales", fecha: "2026-08-04", horas: 4 }
        ];

        // Instancias de Chart.js
        let chartCursos, chartDocentes;

        // Configuración global de Chart.js
        Chart.defaults.font.family = "'Inter', sans-serif";
        Chart.defaults.color = '#64748b'; // slate-500

        // --- 2. Funciones de Interfaz (Modal) ---
        function abrirModal() {
            document.getElementById('modal-agregar').classList.remove('hidden');
            document.getElementById('form-registro').reset();
        }

        function cerrarModal() {
            document.getElementById('modal-agregar').classList.add('hidden');
        }

        // --- 3. Funciones de Lógica (CRUD) ---
        document.getElementById('form-registro').addEventListener('submit', function(e) {
            e.preventDefault();
            
            // Obtener datos del formulario
            const nuevoRegistro = {
                id: Date.now(), // ID único simple
                docente: document.getElementById('input-docente').value,
                curso: document.getElementById('input-curso').value,
                fecha: document.getElementById('input-fecha').value,
                horas: parseInt(document.getElementById('input-horas').value)
            };

            // Agregar al estado
            registrosLaboratorio.push(nuevoRegistro);
            
            // Actualizar UI
            actualizarDashboard();
            cerrarModal();
        });

        function eliminarRegistro(id) {
            registrosLaboratorio = registrosLaboratorio.filter(reg => reg.id !== id);
            actualizarDashboard();
        }

        // --- 4. Renderizado ---
        function actualizarDashboard() {
            renderTable();
            actualizarKPIs();
            actualizarGraficos();
        }

        function renderTable() {
            const tbody = document.getElementById('tabla-registros');
            const emptyState = document.getElementById('empty-state');
            
            tbody.innerHTML = '';
            
            if (registrosLaboratorio.length === 0) {
                emptyState.classList.remove('hidden');
            } else {
                emptyState.classList.add('hidden');
                
                registrosLaboratorio.forEach(item => {
                    const tr = document.createElement('tr');
                    tr.className = "hover:bg-slate-50 transition-colors group";
                    
                    // Extraer inicial del docente
                    const partesNombre = item.docente.split(' ');
                    const apellido = partesNombre.length > 2 ? partesNombre[partesNombre.length - 2] : partesNombre[1];
                    const inicial = apellido ? apellido.charAt(0) : item.docente.charAt(0);
                    
                    // Formatear la fecha (de YYYY-MM-DD a DD/MM/YYYY)
                    let fechaFormateada = item.fecha;
                    if(item.fecha && item.fecha.includes('-')) {
                        const [year, month, day] = item.fecha.split('-');
                        fechaFormateada = `${day}/${month}/${year}`;
                    }

                    tr.innerHTML = `
                        <td class="py-3 px-6 font-medium text-slate-800 border-l-4" style="border-left-color: ${COLORES_CURSOS[item.curso].border}">${item.curso}</td>
                        <td class="py-3 px-6">
                            <div class="flex items-center gap-3">
                                <div class="w-7 h-7 rounded-full bg-slate-200 flex items-center justify-center text-xs font-bold text-slate-600" style="background-color: ${COLORES_DOCENTES[item.docente]}20; color: ${COLORES_DOCENTES[item.docente]}">
                                    ${inicial}
                                </div>
                                <span class="text-slate-700">${item.docente}</span>
                            </div>
                        </td>
                        <td class="py-3 px-6 text-slate-600 font-medium text-sm">
                            <div class="flex items-center gap-1.5">
                                <i class="ph ph-calendar-blank text-slate-400 text-lg"></i>
                                ${fechaFormateada}
                            </div>
                        </td>
                        <td class="py-3 px-6 text-center font-bold text-slate-700">
                            <span class="bg-slate-100 px-3 py-1 rounded-full">${item.horas} h</span>
                        </td>
                        <td class="py-3 px-6 text-center">
                            <button onclick="eliminarRegistro(${item.id})" class="text-red-400 hover:text-red-600 hover:bg-red-50 p-2 rounded-lg transition-colors opacity-0 group-hover:opacity-100 focus:opacity-100" title="Eliminar registro">
                                <i class="ph ph-trash text-lg"></i>
                            </button>
                        </td>
                    `;
                    tbody.appendChild(tr);
                });
            }
        }

        function actualizarKPIs() {
            // Calcular total de horas
            const totalHoras = registrosLaboratorio.reduce((sum, item) => sum + item.horas, 0);
            document.getElementById('kpi-total-horas').textContent = totalHoras;

            // Calcular cursos únicos registrados
            const cursosUnicos = new Set(registrosLaboratorio.map(item => item.curso)).size;
            document.getElementById('kpi-total-cursos').textContent = cursosUnicos;

            // Encontrar el curso con más horas en total
            const horasPorCurso = agruparPor('curso');
            let cursoTop = '-';
            let maxHoras = 0;
            
            for (const [curso, horas] of Object.entries(horasPorCurso)) {
                if (horas > maxHoras) {
                    maxHoras = horas;
                    cursoTop = curso;
                }
            }

            document.getElementById('kpi-curso-top').textContent = cursoTop;
            document.getElementById('kpi-curso-top-horas').textContent = maxHoras > 0 ? `${maxHoras} hrs en total` : '0 hrs';
        }

        // Función utilitaria para agrupar horas por propiedad (curso o docente)
        function agruparPor(propiedad) {
            return registrosLaboratorio.reduce((acc, item) => {
                acc[item[propiedad]] = (acc[item[propiedad]] || 0) + item.horas;
                return acc;
            }, {});
        }

        function actualizarGraficos() {
            const horasPorCurso = agruparPor('curso');
            const horasPorDocente = agruparPor('docente');

            // --- Actualizar Gráfico de Cursos ---
            const labelsCursos = Object.keys(horasPorCurso);
            const dataCursos = Object.values(horasPorCurso);
            const bgCursos = labelsCursos.map(c => COLORES_CURSOS[c].bg);
            const borderCursos = labelsCursos.map(c => COLORES_CURSOS[c].border);

            if (chartCursos) {
                chartCursos.data.labels = labelsCursos;
                chartCursos.data.datasets[0].data = dataCursos;
                chartCursos.data.datasets[0].backgroundColor = bgCursos;
                chartCursos.data.datasets[0].borderColor = borderCursos;
                chartCursos.update();
            } else {
                const ctxCursos = document.getElementById('chartCursos').getContext('2d');
                chartCursos = new Chart(ctxCursos, {
                    type: 'bar',
                    data: {
                        labels: labelsCursos,
                        datasets: [{
                            label: 'Horas Totales',
                            data: dataCursos,
                            backgroundColor: bgCursos,
                            borderColor: borderCursos,
                            borderWidth: 1,
                            borderRadius: 6,
                            barThickness: 'flex',
                            maxBarThickness: 50
                        }]
                    },
                    options: {
                        responsive: true,
                        maintainAspectRatio: false,
                        plugins: {
                            legend: { display: false }
                        },
                        scales: {
                            y: {
                                beginAtZero: true,
                                grid: { borderDash: [4, 4], color: '#e2e8f0' }
                            },
                            x: {
                                grid: { display: false }
                            }
                        }
                    }
                });
            }

            // --- Actualizar Gráfico de Docentes ---
            const labelsDocentes = Object.keys(horasPorDocente);
            const dataDocentes = Object.values(horasPorDocente);
            const coloresDocentesList = labelsDocentes.map(d => COLORES_DOCENTES[d]);

            if (chartDocentes) {
                chartDocentes.data.labels = labelsDocentes;
                chartDocentes.data.datasets[0].data = dataDocentes;
                chartDocentes.data.datasets[0].backgroundColor = coloresDocentesList;
                chartDocentes.update();
            } else {
                const ctxDocentes = document.getElementById('chartDocentes').getContext('2d');
                chartDocentes = new Chart(ctxDocentes, {
                    type: 'doughnut',
                    data: {
                        labels: labelsDocentes,
                        datasets: [{
                            data: dataDocentes,
                            backgroundColor: coloresDocentesList,
                            borderWidth: 2,
                            borderColor: '#ffffff',
                            hoverOffset: 4
                        }]
                    },
                    options: {
                        responsive: true,
                        maintainAspectRatio: false,
                        cutout: '70%',
                        plugins: {
                            legend: { display: false } // Usamos leyenda HTML personalizada
                        }
                    }
                });
            }

            // Actualizar leyenda HTML de docentes
            const legendContainer = document.getElementById('docentes-legend');
            legendContainer.innerHTML = '';
            labelsDocentes.forEach(docente => {
                const div = document.createElement('div');
                div.className = "flex items-center justify-between p-2 rounded bg-slate-50";
                
                // Formatear nombre corto
                const partes = docente.split(' ');
                const nombreCorto = partes.length > 2 ? `${partes[0]} ${partes[partes.length - 2]} ${partes[partes.length - 1]}` : docente;

                div.innerHTML = `
                    <div class="flex items-center gap-2 truncate">
                        <span class="w-3 h-3 rounded-full flex-shrink-0" style="background-color: ${COLORES_DOCENTES[docente]}"></span>
                        <span class="truncate font-medium text-slate-700" title="${docente}">${nombreCorto}</span>
                    </div>
                    <span class="font-bold text-slate-800">${horasPorDocente[docente]}h</span>
                `;
                legendContainer.appendChild(div);
            });
        }

        // Inicializar el Dashboard al cargar la página
        actualizarDashboard();

    </script>
